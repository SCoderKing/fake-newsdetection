function loadDashboard() { const checks = localStorage.getItem("newsChecks") || 0; const quizzes = localStorage.getItem("quizAttempts") || 0; const surveys = localStorage.getItem("surveyResponses") || 0; const score = localStorage.getItem("mediaLiteracyScore"); document.getElementById("checkCount").textContent = checks; document.getElementById("quizCount").textContent = quizzes; document.getElementById("surveyCount").textContent = surveys; document.getElementById("awarenessScore").textContent = score !== null ? score + " / 10" : "0 / 10"; } loadDashboard();
